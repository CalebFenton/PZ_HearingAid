module HearingAid
{
    imports
    {
        Base
    }

    /*
     * Display names come from Translate/<LANG>/Recipes.json, keyed by recipe name.
     * Callbacks are in lua/shared/HearingAid/HearingAidRecipes.lua, which also replaces each
     * SkillRequired with the matching *SkillLevel sandbox option; keep these equal to its default.
     *
     * Repairs and upgrades are fine work: they need a table and light, tweezers, and reading
     * glasses or a magnifier. An input line accepts its listed items and its tagged items alike.
     */

    craftRecipe RepairHearingAid
    {
        timedAction = MakingElectrical,
        time = 100,
        Tags = AnySurfaceCraft,
        category = Electrical,
        SkillRequired = Electricity:2,
        xpAward = Electricity:10,
        AllowBatchCraft = false,
        OnCreate = HearingAid.Recipes.transferState,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1;MayDegradeLight],
            item 1 tags[base:tweezers] mode:keep,
            item 1 [Base.Glasses_Reading] tags[base:magnifier] mode:keep,
            item 1 [HearingAid.BrokenHearingAid] mode:destroy flags[IsNotWorn],
            item 1 [Base.Earbuds],
            item 1 [Base.AlcoholWipes;Base.AlcoholedCottonBalls],
            item 1 [Base.Glue],
        }
        outputs
        {
            item 1 HearingAid.HearingAid,
        }
    }

    craftRecipe OptimizeHearingAid
    {
        timedAction = MakingElectrical,
        time = 120,
        Tags = AnySurfaceCraft,
        category = Electrical,
        SkillRequired = Electricity:4,
        xpAward = Electricity:15,
        AllowBatchCraft = false,
        OnCreate = HearingAid.Recipes.transferState,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1;MayDegradeLight],
            item 1 tags[base:tweezers] mode:keep,
            item 1 [Base.Glasses_Reading] tags[base:magnifier] mode:keep,
            item 1 [HearingAid.HearingAid] mode:destroy flags[IsNotWorn],
            item 1 [Base.WristWatch_Left_DigitalBlack;Base.WristWatch_Right_DigitalBlack;Base.WristWatch_Left_DigitalRed;Base.WristWatch_Right_DigitalRed;Base.WristWatch_Left_DigitalDress;Base.WristWatch_Right_DigitalDress] flags[IsNotWorn],
            item 1 [Base.ElectricWire],
            item 1 [Base.Glue],
        }
        outputs
        {
            item 1 HearingAid.EfficientHearingAid,
        }
    }

    craftRecipe BoostHearingAid
    {
        timedAction = MakingElectrical,
        time = 150,
        Tags = AnySurfaceCraft,
        category = Electrical,
        SkillRequired = Electricity:8,
        xpAward = Electricity:25,
        AllowBatchCraft = false,
        OnTest = HearingAid_isBoostEnabled,
        OnAddToMenu = HearingAid_isBoostEnabled,
        OnCreate = HearingAid.Recipes.transferState,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep flags[Prop1;MayDegradeLight],
            item 1 tags[base:tweezers] mode:keep,
            item 1 [Base.Glasses_Reading] tags[base:magnifier] mode:keep,
            item 1 [Base.Scalpel] mode:keep flags[MayDegradeVeryLight],
            item 1 [HearingAid.EfficientHearingAid] mode:destroy flags[IsNotWorn],
            item 1 [Base.Amplifier],
            item 1 [Base.Microphone],
            item 2 [Base.ElectricWire],
            item 1 tags[base:epoxy],
        }
        outputs
        {
            item 1 HearingAid.BoostedHearingAid,
        }
    }

    craftRecipe DismantleHearingAid
    {
        timedAction = DismantleElectrical,
        time = 30,

        category = Electrical,
        xpAward = Electricity:2,
        OnCreate = HearingAid.Recipes.returnBattery,
        inputs
        {
            item 1 tags[base:screwdriver] mode:keep,
            item 1 [HearingAid.BrokenHearingAid;HearingAid.HearingAid;HearingAid.EfficientHearingAid;HearingAid.BoostedHearingAid] mode:destroy flags[IsNotWorn;ItemCount],
        }

        }
    }
}
